01 · text files

FILE I/O — fopen & fclose

file_io.c
#include <stdio.h>

int main(void)
{
    FILE *fp;

    /* open for writing */
    fp = fopen("data.txt", "w");
    if (fp == NULL) {
        printf("Error opening file!\n");
        return (1);
    }

    fprintf(fp, "Score: %d\n", 95);
    fprintf(fp, "Name: %s\n", "Alice");
    fclose(fp);

    /* open for reading */
    fp = fopen("data.txt", "r");
    char line[100];
    while (fgets(line, 100, fp) != NULL)
        printf("%s", line);
    fclose(fp);

    return (0);
}
ModePurposeCreates?
"r"read textNo — error if missing
"w"write textYes — truncates if exists
"a"append textYes — adds to end
"rb"read binaryNo
"wb"write binaryYes — truncates
⚠ ALWAYS CHECK fopen
fopen returns NULL on failure (file not found, permission denied). Using a NULL file pointer crashes the program.
ALWAYS fclose
Unflushed data may be lost. fclose writes buffers to disk and releases the file handle. One fclose per fopen.
02 · text files

FORMATTED FILE I/O

formatted_io.c
/* fprintf — write formatted data to file */
FILE *out = fopen("grades.txt", "w");

for (int i = 0; i < n; i++)
    fprintf(out, "%s %d %.2f\n",
            names[i], ids[i], gpas[i]);
fclose(out);

/* fscanf — read formatted data from file */
FILE *in = fopen("grades.txt", "r");
char name[50]; int id; double gpa;

while (fscanf(in, "%s %d %lf",
              name, &id, &gpa) == 3)
{
    printf("%-10s %5d %5.2f\n",
           name, id, gpa);
}
fclose(in);

/* fscanf returns number of items read
   or EOF at end of file               */
fprintf = printf TO FILE
Same format strings, same placeholders — but first argument is the FILE * instead of stdout. fprintf(stdout, ...) = printf(...).
fscanf RETURN VALUE
Returns the number of items successfully read. Check == expected_count to detect end-of-file or malformed input.
TEXT FORMAT
Data is stored as human-readable characters. 42 becomes the characters '4' and '2'. Editable with any text editor, but larger than binary.
⚠ fscanf & STRINGS
%s stops at whitespace. For names with spaces, use fgets and parse manually, or use a delimiter format.
03 · binary files

BINARY FILES

binary.c
typedef struct {
    int id; char name[30]; double balance;
} account_t;

/* WRITE array of structs to binary file */
account_t accts[3] = {
    {101, "Alice", 5000.00},
    {102, "Bob",   3200.50},
    {103, "Carol", 8750.25}
};

FILE *fp = fopen("accts.bin", "wb");
fwrite(accts, sizeof(account_t), 3, fp);
fclose(fp);

/* READ back */
account_t buf[3];
fp = fopen("accts.bin", "rb");
fread(buf, sizeof(account_t), 3, fp);
fclose(fp);

printf("%s: $%.2f\n", buf[1].name,
       buf[1].balance);  → Bob: $3200.50
fwrite(ptr,sz,n,fp)
Write n items, each sz bytes, from memory at ptr to file fp.
fread(ptr,sz,n,fp)
Read n items of sz bytes from file into memory. Returns number of items actually read.
TEXT vs BINARY
Text: human-readable, portable, larger. Binary: exact memory image, compact, fast, but not human-readable and platform-dependent.
⚠ PORTABILITY
Binary files depend on struct layout, byte order, and padding. A binary file written on one machine may not read correctly on another with different architecture.
04 · binary files

RANDOM ACCESS

random_access.c
/* jump to specific record in binary file */
FILE *fp = fopen("accts.bin", "rb");

/* read the 3rd record (index 2) */
fseek(fp, 2 * sizeof(account_t), SEEK_SET);
account_t rec;
fread(&rec, sizeof(account_t), 1, fp);

/* ftell — current position in bytes */
long pos = ftell(fp);
printf("Position: %ld bytes\n", pos);

/* rewind — go back to start */
rewind(fp);  /* same as fseek(fp,0,SEEK_SET) */

/* fseek origins:
   SEEK_SET  — from beginning
   SEEK_CUR  — from current position
   SEEK_END  — from end of file       */

/* modify record in place */
fp = fopen("accts.bin", "r+b");
fseek(fp, 1 * sizeof(account_t), SEEK_SET);
rec.balance = 9999.99;
fwrite(&rec, sizeof(account_t), 1, fp);
fclose(fp);
DIRECT ACCESS
With fixed-size records, fseek(fp, index * sizeof(record), SEEK_SET) jumps directly to any record — O(1) access, no scanning.
UPDATE IN PLACE
Open with "r+b", seek to the record, write the new data. The file size doesn't change — just the bytes at that position.
FILE POSITION
ftell(fp) returns the current byte offset. After reading one record, it advances by sizeof(record) bytes.
⚠ TEXT vs BINARY SEEK
fseek with byte offsets only works reliably on binary files. In text mode, line endings may be translated, making byte counts unreliable.
05 · database

SEARCHING A DATABASE

db_search.c
/* sequential search in a binary file */
int find_account(FILE *fp, int target_id,
                 account_t *result)
{
    account_t rec;
    rewind(fp);

    while (fread(&rec, sizeof(account_t),
                 1, fp) == 1)
    {
        if (rec.id == target_id) {
            *result = rec;
            return 1;    ← found
        }
    }
    return 0;              ← not found
}

/* usage */
account_t a;
if (find_account(fp, 102, &a))
    printf("%s: $%.2f\n", a.name, a.balance);
else
    printf("Account not found\n");
SEQUENTIAL FILE SEARCH
Read records one by one from the file until the target is found or EOF is reached. O(n) — must check every record in the worst case.
BINARY SEARCH ON FILE
If the file is sorted by key, use fseek to jump to the middle record, compare, and halve the search space — O(log n) disk reads.
LOAD vs STREAM
Small DB: load entire file into an array, search in memory. Large DB: stream from disk, process one record at a time to save RAM.
⚠ FILE MUST BE OPEN
The FILE * must remain open during the search. Don't close and reopen between reads — use rewind to restart.
06 · searching

LINEAR SEARCH — ANALYSIS

linear_search.c
int linear_search(int arr[], int n,
                  int target)
{
    for (int i = 0; i < n; i++)
        if (arr[i] == target)
            return i;
    return -1;
}
CaseComparisonsComplexity
Best1 (found at [0])O(1)
Averagen/2O(n)
Worstn (not found)O(n)
WHEN TO USE
Unsorted data, small arrays, or when you need to search once. No preprocessing required.
SENTINEL VARIANT
Place the target at arr[n] as a sentinel — eliminates the i < n check each iteration. Slightly faster inner loop, same O(n).
SCALING
1,000 items: ~500 avg comparisons. 1,000,000 items: ~500,000. For large sorted data, binary search is orders of magnitude faster.
07 · searching

BINARY SEARCH — ANALYSIS

binary_search.c
int binary_search(int arr[], int n,
                  int target)
{
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) lo = mid+1;
        else hi = mid-1;
    }
    return -1;
}
nMax comparisons
104
1,00010
1,000,00020
1 billion30
O(log₂ n)
Each comparison eliminates half the remaining elements. Doubling the array size adds only ONE more comparison.
OVERFLOW-SAFE MID
mid = lo + (hi - lo) / 2 avoids integer overflow that (lo + hi) / 2 can cause when both are large.
⚠ MUST BE SORTED
Binary search on an unsorted array is undefined — it may return wrong results or miss existing elements. Sort first (one-time O(n log n) cost).
08 · sorting

BUBBLE SORT — ANALYSIS

bubble_sort.c — optimized
void bubble_sort(int arr[], int n)
{
    for (int i = 0; i < n-1; i++) {
        int swapped = 0;
        for (int j = 0; j < n-1-i; j++) {
            if (arr[j] > arr[j+1]) {
                int t = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = t;
                swapped = 1;
            }
        }
        if (!swapped) break; ← early exit
    }
}

/* trace: {4,2,3,1}
pass 1: 2,3,1,|4|  (3 swaps)
pass 2: 2,1,|3|    (1 swap)
pass 3: 1,|2|      (1 swap)  done ✓ */
CaseComparisonsSwaps
Best (sorted)n-10
Averagen²/4n²/4
Worst (reverse)n²/2n²/2
OPTIMIZATION
The swapped flag detects if no swaps occurred — array is already sorted. Best case drops from O(n²) to O(n).
VERDICT
Simple to understand and code. Too slow for large data. Good for educational purposes and very small arrays.
09 · sorting

INSERTION SORT — ANALYSIS

insertion_sort.c
void insertion_sort(int arr[], int n)
{
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j+1] = arr[j];
            j--;
        }
        arr[j+1] = key;
    }
}

/* trace: {4,2,3,1}
i=1: key=2, shift 4→     [2,4,3,1]
i=2: key=3, shift 4→     [2,3,4,1]
i=3: key=1, shift 4,3,2→ [1,2,3,4] ✓ */
CaseComparisonsShifts
Best (sorted)n-10
Averagen²/4n²/4
Worst (reverse)n²/2n²/2
ADAPTIVE
O(n) for nearly sorted data — few elements are far from their correct position, so few shifts needed. The best simple sort for "almost sorted" input.
ONLINE
Can sort data as it arrives — no need to know the total count upfront. Each new element is inserted into the correct position immediately.
VERDICT
Best of the O(n²) sorts for practical use. Preferred for small arrays (n < 50) and nearly-sorted data. Used as a subroutine in advanced sorts.
10 · sorting

SELECTION SORT — ANALYSIS

selection_sort.c
void selection_sort(int arr[], int n)
{
    for (int i = 0; i < n-1; i++) {
        int min = i;
        for (int j = i+1; j < n; j++)
            if (arr[j] < arr[min])
                min = j;
        if (min != i) {
            int t = arr[i];
            arr[i] = arr[min];
            arr[min] = t;
        }
    }
}

/* trace: {4,2,3,1}
i=0: min=1@[3] swap→ [1,2,3,4]
i=1: min=2@[1] no-op
i=2: min=3@[2] no-op  done ✓ */
CaseComparisonsSwaps
Bestn²/20
Averagen²/2n/2
Worstn²/2n-1
FEWEST SWAPS
At most n-1 swaps — one per outer pass. Best choice when swaps are expensive (e.g., large records, slow memory).
CONSISTENT
Always does n²/2 comparisons, regardless of input order. No best-case speedup like insertion sort — but predictable performance.
VERDICT
Simple, minimal memory writes. Not adaptive. Use when write cost dominates or for very small arrays.
11 · comparison

SORTING COMPARISON

Algorithm Best Average Worst Stable? Swaps
Bubble O(n) O(n²) O(n²) Yes Many
Insertion O(n) O(n²) O(n²) Yes Moderate
Selection O(n²) O(n²) O(n²) No Fewest
SPACE COMPLEXITY
All three are in-place — O(1) extra memory. They sort the array directly, no auxiliary array needed.
CHOOSE WISELY
Nearly sorted? → Insertion. Expensive swaps? → Selection. Teaching? → Bubble. Large data? → Use qsort() from stdlib.h (O(n log n)).
STABLE SORT
A sort is stable if equal elements keep their original order. Important when sorting by multiple criteria (e.g., sort by name, then by grade).
THE BIG PICTURE
These three are introductory sorts. Real-world programs use O(n log n) algorithms like quicksort, mergesort, or heapsort. But understanding O(n²) sorts builds the foundation.
