    Logic Error

    Type Casting — Converting Between Types

    Sometimes you need to convert one data type to another

    🔄 Implicit Casting (Automatic)
    C automatically converts smaller types to larger types. This is called widening.
    implicit.c
    int a = 5;
float b = a;  // int → float (automatic)
// b is now 5.000000

int x = 5, y = 2;
float result = x / y;  // ⚠️ Still 2.0!
// Integer division happens FIRST, then converts
    🎯 Explicit Casting (You force it)
    You manually convert using (type) syntax. Use when you need precision.
    explicit.c
    int x = 5, y = 2;
float result = (float)x / y;  // 2.500000 ✓
// Cast x to float BEFORE division

float pi = 3.14159;
int truncated = (int)pi;  // 3 (decimal lost!)
    🚫 Common Bug: float avg = marks / total; — If both are int, you get integer division! Fix: float avg = (float)marks / total;

    ⚠️ Integer Division

    This catches EVERY beginner — pay close attention!

    When you divide two integers, C throws away the decimal part!
    printf("%d\n", 5 / 2);     // 2  (not 2.5!)
printf("%d\n", 7 / 3);     // 2  (not 2.33!)
printf("%d\n", 1 / 2);     // 0  (not 0.5!)
    🚫 The Fix: Use float or 5.0 / 2.0
    printf("%f\n", 5.0 / 2.0);  // 2.500000 ✓
printf("%f\n", (float)5 / 2); // 2.500000 ✓
    🌍 Analogy
    Dividing 5 cookies among 2 people:
    Integer: Each gets 2 cookies, 1 is left over → 5/2=2, 5%2=1
    Float: Each gets 2.5 cookies → 5.0/2.0=2.5
    🔢 Modulus = Remainder
    17 % 5 = 2 because 17 = 5 × 3 + 2

    Real use: Check if a number is even: n % 2 == 0

    Increment & Decrement

    Pre vs Post
    ++a → increment FIRST, then use the value
    a++ → use the value FIRST, then increment
    #include <stdio.h>

int main()
{
    int a = 5;
    printf("%d\n", a++);  // prints 5, then a=6
    printf("%d\n", a);    // prints 6

    int b = 5;
    printf("%d\n", ++b);  // b=6 first, prints 6
    printf("%d\n", b);    // prints 6

    return 0;
}
    ▶ Output
    5 6 6 6
    ExpressionBeforePrintedAfter
    a++556
    ++b566
    a--665
    --b655
    🌍 Simple Rule
    Post (a++): "Use me NOW, change me LATER"
    Pre (++a): "Change me FIRST, then use me"
    💡 When used alone (like a++; on its own line), pre and post are identical. The difference only matters inside expressions.

    Program: Prime Number Check

    📋 What is Prime?
    A number > 1 that has exactly two factors: 1 and itself.
    Examples: 2, 3, 5, 7, 11, 13, 17, 19...
    Not prime: 4 (1,2,4), 6 (1,2,3,6), 9 (1,3,9)
    prime.c
    #include <stdio.h>

int main()
{
    int n, isPrime = 1;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n <= 1)
        isPrime = 0;

    for (int i = 2; i * i <= n; i++)
    {
        if (n % i == 0)
        {
            isPrime = 0;
            break;
        }
    }

    if (isPrime)
        printf("%d is Prime\n", n);
    else
        printf("%d is Not Prime\n", n);

    return 0;
}
    ▶ Output
    Enter a number: 29 29 is Prime
    🔍 Algorithm
    For n=29, check divisibility by 2,3,4,5 (√29≈5.4):
    29 % 2 = 1 (not divisible) ↓
    29 % 3 = 2 (not divisible) ↓
    29 % 4 = 1 (not divisible) ↓
    29 % 5 = 4 (not divisible) ↓
    No divisor found → Prime!
    💡 Optimization: We only check up to √n. If a number has a factor > √n, it must also have one < √n.

    Halves search space each step — REQUIRES A SORTED ARRAY!

    binary_search.c

int binarySearch(int arr[], int n, int key)
{
    int low = 0, high = n - 1;

    while (low <= high)
    {
        int mid = low + (high - low) / 2;

        if (arr[mid] == key)
            return mid;
        else if (key < arr[mid])
            high = mid - 1; // Discard right half
        else
            low = mid + 1;  // Discard left half
    }
    return -1;
}

    n3->data = 30; n3->next = NULL;

    printList(n1);

    free(n3); free(n2); free(n1);
    return 0;
}
    ▶ Output
    List: 10 -> 20 -> 30 -> NULL
    ⚠️ Never Move Head!
    Always use a temporary pointer: Node *temp = head;
    If you advance head = head->next during traversal, you lose access to the list forever!

    Insert at Beginning: O(1) Time

    Instant insertion using double pointers (Node** head)

    insert_head.c
    #include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

void insertAtHead(Node **head, int val)
{
    Node *newNode = (Node*)malloc(sizeof(Node));
    newNode->data = val;
    newNode->next = *head;  // Point new node to old head
    *head = newNode;        // Update head to new node!
}

void printList(Node *head) {
    while (head) { printf("%d -> ", head->data); head = head->next; }
    printf("NULL
");
}

int main()
{
    Node *head = NULL; // Empty list

    insertAtHead(&head, 30);
    insertAtHead(&head, 20);
    insertAtHead(&head, 10);

    printList(head);
    return 0;
}
    ▶ Output
    10 -> 20 -> 30 -> NULL
    ⚡ Why O(1)?
    No shifting required! Only 2 pointer assignments:
    1. newNode->next = *head;
    2. *head = newNode;

    Unlike arrays which take O(n) to insert at index 0, Linked Lists perform this in instant O(1) time!

    Deleting a Node: Pointer Bypass

    Unlinking target node from chain and freeing heap memory

    delete_node.c
    #include <stdio.h>
#include <stdlib.h>

typedef struct Node { int data; struct Node *next; } Node;

void deleteNode(Node **head, int key)
{
    Node *temp = *head, *prev = NULL;

    // Case 1: Head node holds key
    if (temp != NULL && temp->data == key) {
        *head = temp->next; // Change head
        free(temp);         // Free old head
        return;
    }

    // Case 2: Search for key
    while (temp != NULL && temp->data != key) {
        prev = temp;
        temp = temp->next;
    }

    if (temp == NULL) return; // Key not found!

    // Bypass node & free
    prev->next = temp->next;
    free(temp);
}
    ✂️ Pointer Bypass Concept
    To delete node B in A -> B -> C:

    Set prev->next = temp->next; (A now points directly to C).
    Then call free(temp); to reclaim B's RAM!

    Array-Based Stack Implementation

    stack_array.c
    #include <stdio.h>
#define MAX 5

typedef struct {
    int items[MAX];
    int top;
} Stack;

void init(Stack *s) { s->top = -1; }

void push(Stack *s, int val) {
    if (s->top == MAX - 1) { printf("Stack Overflow!
"); return; }
    s->items[++(s->top)] = val;
}

int pop(Stack *s) {
    if (s->top == -1) { printf("Stack Underflow!
"); return -1; }
    return s->items[(s->top)--];
}

int main()
{
    Stack s; init(&s);
    push(&s, 10); push(&s, 20); push(&s, 30);

    printf("Popped: %d
", pop(&s));
    printf("Popped: %d
", pop(&s));
    return 0;
}
    ▶ Output
    Popped: 30 Popped: 20
    🔍 The Top Pointer
    top = -1 means stack is empty.
    push() increments ++top and stores data.
    pop() returns data and decrements top--.

typedef struct TreeNode { int val; struct TreeNode *left, *right; } TreeNode;

void inorder(TreeNode *root) {
    if (root == NULL) return;
    inorder(root->left);          // 1. Left
    printf("%d ", root->val);     // 2. Root
    inorder(root->right);         // 3. Right

void preorder(TreeNode *root) {
    if (root == NULL) return;
    printf("%d ", root->val);     // 1. Root

    preorder(root->right);        // 3. Right

    if (root == NULL) return;
    postorder(root->left);        // 1. Left
    postorder(root->right);       // 2. Right
    printf("%d ", root->val);     // 3. Root
}

    Merge Sort: Divide, Conquer & Combine

    Guaranteed O(n log n) stable sorting algorithm

    merge_sort.c
    #include <stdio.h>

void merge(int arr[], int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m;
    int L[n1], R[n2];
    for (int i = 0; i < n1; i++) L[i] = arr[l + i];
    for (int j = 0; j < n2; j++) R[j] = arr[m + 1 + j];

    int i = 0, j = 0, k = l;
    while (i < n1 && j < n2)
        arr[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
}

void mergeSort(int arr[], int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(arr, l, m);     // Divide left
        mergeSort(arr, m + 1, r); // Divide right
        merge(arr, l, m, r);      // Combine
    }
}
    📊 Why O(n log n)?
    • Array is divided in half log₂ n times.
    • Each level takes O(n) to merge.
    • Total: O(n log n) in ALL cases (Best, Average, Worst)!
